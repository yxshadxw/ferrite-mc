# Fallbacks for 26.3 (777) content that 26.2 (776) does not have.
#
#   <registries> <pattern> -> <target>
#
# `registries` is a comma-separated list of registry names without the `minecraft:`
# namespace; `block` also drives the block state remap (properties are carried over where
# the target block has them, the rest take the target's default). `*` captures any run of
# characters and is substituted into the target. The first matching rule wins. Anything
# with no rule maps to the registry default (air, pig, ...) in packets, and is dropped from
# tags.
#
# Rules prefer the same collision shape over the same look, so that client-side movement
# prediction agrees with the server.

# Concrete and wool slabs/stairs (26.3).
block,item minecraft:*_concrete_slab -> minecraft:smooth_stone_slab
block,item minecraft:*_concrete_stairs -> minecraft:stone_stairs
block,item minecraft:*_wool_slab -> minecraft:smooth_stone_slab
block,item minecraft:*_wool_stairs -> minecraft:stone_stairs

# Poplar wood set -> birch (identical property sets).
block,item minecraft:orange_poplar_leaves -> minecraft:birch_leaves
block,item minecraft:red_poplar_leaves -> minecraft:birch_leaves
block,item minecraft:yellow_poplar_leaves -> minecraft:birch_leaves
block minecraft:potted_poplar_sapling -> minecraft:potted_birch_sapling
block,item minecraft:stripped_poplar_* -> minecraft:stripped_birch_*
block,item,entity_type minecraft:poplar_* -> minecraft:birch_*

# Other new blocks.
block,item minecraft:red_shrub -> minecraft:bush
block minecraft:shelf_mushroom -> minecraft:cocoa
item minecraft:shelf_mushroom -> minecraft:cocoa_beans
block,item minecraft:straw_bed -> minecraft:white_bed

# New items without a block.
item minecraft:*_cushion -> minecraft:*_carpet
item minecraft:*_map -> minecraft:map

# New entity types.
entity_type minecraft:cushion -> minecraft:interaction

# New particles.
particle_type minecraft:*_poplar_leaves -> minecraft:cherry_leaves

# Data-driven registries (resolved at runtime against the entries the core sends).
worldgen/biome minecraft:dappled_forest -> minecraft:forest
